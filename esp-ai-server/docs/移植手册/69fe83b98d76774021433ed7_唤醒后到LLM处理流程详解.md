# ESP-AI Server 唤醒设备后处理流程详解

## 📋 概述

本文档详细说明当用户唤醒 ESP32 设备后，服务端从接收到唤醒信号到启动 LLM 对话处理的完整流程。

**版本**: v2.82.70


**处理逻辑**:

```python
elif t == "start":
    # 场景1: 已在ASR状态但未在TTS播放 → 取消并重启
    if fsm.get() == SessionState.ASR and not session.tts_playing:
        info("[Session] 已在 ASR 状态（可能来自外部唤醒），取消并重启")
        session.cancel_event.set()
        await asyncio.sleep(0.05)
        session.cancel_event.clear()
        session.stop_asr()

    # 场景2: 已在ASR状态且正在TTS播放 → 忽略重复start

    # 场景3: 正常启动流程
    info("[收到] start 命令")

    # 3.1 如果正在播放TTS，先打断

        info("[Session] 打断 TTS")
        if tts_done_waiter and not tts_done_waiter.done():
            tts_done_waiter.cancel()
        await session.interrupt()
        try:
            await asyncio.wait_for(session.tts_playback_done.wait(), timeout=0.5)
        except asyncio.TimeoutError:
            pass
    else:

            tts_done_waiter.cancel()

    # 3.2 发送会话开始消息
    await channel.send_json({"type": "session_start", "session_id": "0001"})

    # 3.3 重置事件
    session.tts_playback_done.set()
    session.tts_audio_ended.set()
    session.tts_playback_done.clear()
    session.tts_audio_ended.clear()

    # 3.4 取消之前的ASR预连接
    if asr_client:
        session.cancel_pre_asr()
        asyncio.create_task(session.pre_connect_asr())

    # 3.5 启动唤醒流程或直接ASR
    if ENABLE_WAKEUP_AUDIO:
        session._wake_audio_played.clear()
        asyncio.create_task(_do_wake_start())  # 异步执行
    else:
        await fsm.set(SessionState.ASR)
        _start_asr_session()
        await channel.send_json({"type": "session_status", "status": "iat_start"})
```

```python
class WakeAudioManager:
    async def play(self, channel, user_config=None):
        audio_data = await self.ensure_cache(user_config)

        # 发送播放指令
        await channel.send_json({"type": "play_audio", "tts_task_id": SID_TTS})
        await channel.send_json({"type": "session_status", "status": "tts_chunk_start"})

        # 分块发送音频
        _CHUNK_SIZE = 2048
        sent = 0
        while sent < _TOTAL:
            chunk = audio_data[sent : sent + _CHUNK_SIZE]
            sent += len(chunk)
            frame = self._voice_generator.make_tts_frame(SID_TTS, chunk, "00")
            await channel.send_bytes(frame)
            await asyncio.sleep(0.02)

        # 发送结束帧
        await channel.send_bytes(self._voice_generator.make_end_frame(SID_TTS))
        await channel.send_json({"type": "session_status", "status": "tts_real_end"})

        # 标记唤醒音频已播放
        session._wake_audio_played.set()
```

```python
async def start_asr(self, on_text, on_vad_end):
    # 4.1 取消之前的ASR任务
    if self.runtime.asr_task and not self.runtime.asr_task.done():
        info("[Session] 检测到旧的 ASR 任务，先行取消")
        self.runtime.asr_task.cancel()

    # 4.2 重置运行时状态
    self.runtime.reset()
    self.audio_processor.reset()
    self.runtime.asr_start_time = time.time()

    # 4.3 创建音频队列
    if not self.runtime.audio_queue:
        self.runtime.audio_queue = asyncio.Queue()

    # 4.4 获取预连接的WebSocket（如果有）
    pre_ws = None
    pre_wrapper = None
    if self.asr_client:
        try:
            pre_ws, pre_wrapper = self.asr_client.take_pre_ws()
        except Exception as e:
            debug(f"[Session] 取走预连接 WS 异常: {e}")

    # 4.5 启动ASR流式识别任务
    self.runtime.asr_task = asyncio.create_task(
        _asr_streaming_loop(
            self.asr_client,
            self.runtime.audio_queue,
            on_text,           # 文本回调
            _vad_cb,          # VAD结束回调
            self.runtime.asr_stop_event,
            pre_ws,           # 预连接WebSocket
            pre_wrapper,
        )
    )

    info("[Session] ASR 流式识别已启动 (pool)")
```

```python
async def _asr_streaming_loop(asr_client, audio_queue, result_callback, vad_end_callback, stop_event, pre_ws=None, pre_wrapper=None):
    # 6.1 建立WebSocket连接
    if pre_ws and time.time() - pre_time < 25:
        ws = pre_ws  # 使用预连接
    else:
        # 建立新连接
        headers = asr_client._get_headers()
        ws = await websockets.connect(asr_client._ws_url, additional_headers=headers)

    # 6.2 初始化连接（某些ASR需要）
    if need_binary_protocol:
        await asr_client.init_connection(ws)

    # 6.3 创建发送和接收任务
    send_task = asyncio.create_task(send_audio())
    recv_task = asyncio.create_task(recv_audio())
    stop_waiter = asyncio.create_task(stop_event.wait())

    # 并行执行
    done, pending = await asyncio.wait([send_task, recv_task, stop_waiter],
                                       return_when=asyncio.FIRST_COMPLETED)
```

**发送音频** (`send_audio` 函数):

```python
async def send_audio():
    while not send_done and not stop_event.is_set():
        try:
            audio_data = await asyncio.wait_for(audio_queue.get(), timeout=0.5)

            if audio_data is None:  # 结束标记
                if ws:
                    await ws.send(b"")  # 发送空字节表示结束
                break

            # 发送音频数据
            await asr_client.send_audio_data(ws, audio_data)

        except asyncio.TimeoutError:
            continue
```

**接收识别结果** (`recv_audio` 函数):

```python
async def recv_audio():
    while not send_done and not stop_event.is_set():
        try:
            response = await asyncio.wait_for(ws.recv(), timeout=30.0)

            # 解析响应
            result = asr_client.parse_response(response)
            text = result.get("text", "")
            is_final = result.get("is_final", False)

            if text:
                # 调用回调函数，更新识别文本
                result_callback(text)

            # 如果是最终结果
            if is_final:
                info("ASR 识别完成")
                # 调用VAD结束回调
                if vad_end_callback and not stop_event.is_set():
                    await vad_end_callback()
                break

        except asyncio.TimeoutError:
            warning("ASR 接收超时")
            break
```

```python
async def on_vad_end():
    # 7.1 防止重复处理
    if session.runtime.asr_processed:
        return
    session.runtime.asr_processed = True

    # 7.2 取消等待的TTS完成回调
    if tts_done_waiter and not tts_done_waiter.done():
        tts_done_waiter.cancel()
        tts_done_waiter = None

    # 7.3 通知设备ASR结束
    await channel.send_json({"type": "session_status", "status": "iat_end"})

    await asyncio.sleep(0.03)

    # 7.4 获取识别文本
    text = session.runtime.asr_full_text
    info(f"ASR 最终识别文本: {text}")

    # 7.5 停止ASR流
    await session.drain_asr()

    # 7.6 如果文本为空，发送会话结束
    if not text.strip():
        await session.send_session_end()
        return

    # 7.7 限流检查
    rate_limit = user_config.rate_limit_rpm if user_config else RATE_LIMIT_MAX_RPM
    if rate_limit > 0:
        now = time.time()
        call_times = [t for t in call_times if now - t < 60]
        if len(call_times) >= rate_limit:
            warning(f"[限流] 设备超过 {rate_limit} 次/分钟限制")
            await session.send_session_end()
            return
        call_times.append(now)

    # 7.8 清除TTS完成事件
    tts_done_event.clear()

    # 7.9 发送识别文本给设备
    try:
        await channel.send_json({"type": "instruct", "command_id": "on_iat_cb", "data": text})
        await asyncio.sleep(0.03)
    except Exception as e:
        warning(f"[VAD] 发送 instruct 失败: {e}")
        return

    # 7.10 启动Pipeline（LLM + TTS）
    asyncio.create_task(session.run_pipeline(text))

    # 7.11 创建TTS完成回调
    async def _on_tts_complete():
        try:
            await tts_done_event.wait()
            if not session._closed:
                await _start_next_asr()
        except asyncio.CancelledError:
            pass
        except Exception as e:
            error(f"[TTS] 完成回调异常: {e}")

    tts_done_waiter = asyncio.create_task(_on_tts_complete())
```

### 步骤8: 启动看门狗（超时控制）

**文件位置**: `app/session/session.py` 第733-778行

**函数**: `session.start_watchdog()`

```python
async def start_watchdog(self, on_vad_end):
    async def _watchdog():
        try:
            max_asr_duration = 30.0  # 最大ASR时长30秒

            while not self.runtime.asr_processed:
                await asyncio.sleep(0.2)

                if self.fsm.get() != SessionState.ASR:
                    continue

                now = time.time()
                lat = self.runtime.asr_last_audio_time  # 最后音频时间
                lr = self.runtime.asr_last_result_time  # 最后识别时间
                st = self.runtime.asr_start_time  # ASR开始时间

                # 超时1: 5秒内未收到音频
                if lr is None and st and now - st > ASR_NO_SPEECH_TIMEOUT:
                    if lat is None:
                        warning(f"[Session] {ASR_NO_SPEECH_TIMEOUT}s 未收到音频")
                    else:
                        warning(f"[Session] {ASR_NO_SPEECH_TIMEOUT}s 无 ASR 识别结果")
                    await self.send_session_end()
                    return

                # 超时2: 收到音频但2秒无识别结果
                if lat and lr is None:
                    if now - lat > ASR_NO_SPEECH_TIMEOUT:
                        warning(f"[Session] 收到音频 {now - lat:.1f}s 但无 ASR 结果")
                        await self.send_session_end()
                        return
                    continue

                # 超时3: 2秒静默触发VAD
                if lr and now - lr > ASR_SILENCE_TIMEOUT and not self.runtime.asr_processed:
                    VAD_TRIGGER.labels(type="silence").inc()
                    await on_vad_end()
                    return

                # 超时4: ASR运行超过30秒强制停止
                if st and now - st > max_asr_duration:
                    warning(f"[Session] ASR 运行超过 {max_asr_duration}s，强制停止")
                    self.stop_asr()
                    return

        except asyncio.CancelledError:
            pass
        except Exception as e:
            error(f"[Session] Watchdog 异常: {e}")

    self._watchdog_task = asyncio.create_task(_watchdog())
```

```python
async def run_pipeline(self, iat_text: str):
    # 9.1 重置所有状态
    self.cancel_event.clear()
    self.tts_playback_done.clear()
    self.tts_drain_ack.clear()
    self.tts_audio_ended.clear()
    self.queues.clear_all()
    self.splitter.reset()

    # 9.2 设置TTS播放标志
    await self.set_tts_playing(True)

    # 9.3 切换状态机
    await self.fsm.set(SessionState.LLM)

    # 9.4 获取LLM处理器
    llm = self.llm_processor

    # 9.5 构建消息
    memory_enabled = LLM_MEMORY_ENABLED
    memory_max = LLM_MEMORY_MAX_MESSAGES
    if self.user_config:
        if self.user_config.llm_memory_enabled is not None:
            memory_enabled = self.user_config.llm_memory_enabled
        if self.user_config.llm_memory_max_messages is not None:
            memory_max = self.user_config.llm_memory_max_messages

    # 是否启用对话记忆
    if memory_enabled:
        self.conversation_memory.max_messages = memory_max
        messages = self.conversation_memory.build_messages(
            llm.system_prompt if hasattr(llm, "system_prompt") else LLM_SYSTEM_PROMPT,
            iat_text,
        )
    else:
        # 无记忆模式
        messages = [
            {"role": "system", "content": llm.system_prompt if hasattr(llm, "system_prompt") else LLM_SYSTEM_PROMPT},
            {"role": "user", "content": iat_text},
        ]

    # 9.6 切换状态为TTS
    await self.fsm.set(SessionState.TTS)

    # 9.7 通知设备开始TTS播放
    await self.channel.send_json({"type": "play_audio", "tts_task_id": SID_TTS})
    await asyncio.sleep(0.03)
    await self.channel.send_json({"type": "session_status", "status": "tts_chunk_start"})

    # 9.8 启动4 Worker Pipeline
    tasks = []
    t_llm = asyncio.create_task(self._llm_task(llm, messages))
    t_splitter = asyncio.create_task(self._splitter_task())
    t_tts = asyncio.create_task(self._tts_task(volc_tts, session))
    t_sender = asyncio.create_task(self._sender_task())
    tasks = [t_llm, t_splitter, t_tts, t_sender]

    info(f"[Session:{self.session_id}] Pipeline 已启动 (4 workers)")
```

```python
async def stream_chat(self, messages: list):
    """流式对话入口"""
    if not self.client:
        yield "LLM 未配置 - 这是模拟响应"
        return

    try:
        info("[OpenAI LLM] 开始对话（支持工具链调用）...")

        tools_param = self._get_tools_param()
        round_num = 0
        failed_tool_calls: set = set()

        while round_num < MAX_TOOL_ROUNDS:
            round_num += 1
            is_first = round_num == 1

            if is_first:
                # 第一轮：流式请求
                response = await self._retry(
                    lambda: self.client.chat.completions.create(
                        model=self.model,
                        messages=messages,
                        stream=True,
                        **({"tools": tools_param} if tools_param else {}),
                    )
                )

                full_text = ""
                chunk_count = 0
                raw_tool_calls: dict[int, dict] = {}

                # 流式接收tokens
                async for chunk in response:
                    delta = chunk.choices[0].delta if chunk.choices else None

                    # 处理推理内容（如果有）
                    if hasattr(delta, "reasoning_content") and delta.reasoning_content:
                        reasoning_content += delta.reasoning_content

                    # 处理工具调用
                    if delta.tool_calls:
                        for tc in delta.tool_calls:
                            # 收集工具调用信息
                            ...

                    # 处理文本内容
                    if delta.content:
                        chunk_count += 1
                        full_text += delta.content
                        yield delta.content  # 实时返回token

                info(f"[OpenAI LLM Pipeline 完成] 共 {chunk_count} 个 chunks")

                # 如果有工具调用，执行工具
                if raw_tool_calls:
                    # 执行每个工具
                    for tc_data in raw_tool_calls.values():
                        tool_result = await self.tool_manager.call_tool(
                            tc_data["function_name"],
                            func_args
                        )
                        # 将结果添加到消息
                        messages.append({
                            "role": "tool",
                            "tool_call_id": tc_data["id"],
                            "content": str(tool_result),
                        })

                    # 继续下一轮（非流式）
                    continue
                else:
                    # 无工具调用，直接返回
                    return

            else:
                # 后续轮次：非流式请求
                response = await self._retry(
                    lambda: self.client.chat.completions.create(
                        model=self.model,
                        messages=messages,
                        stream=False,
                    )
                )
                # 处理响应...

    except Exception as e:
        error(f"[OpenAI LLM] 流式请求异常: {e}")
        yield f"LLM error: {str(e)}"
```

### LLM Task 执行

**文件位置**: `app/session/session.py` 第374-421行

```python
async def _llm_task(self, llm, messages):
    seq_id = 0
    full_text = ""

    try:
        # 流式调用LLM
        async for token in llm.stream_chat(messages):
            # 检查取消信号
            if self.cancel_event.is_set():
                info(f"[Session:{self.session_id}] LLM 收到取消信号")
                break

            # 检查错误
            if token.startswith("LLM error"):
                error(f"[Session:{self.session_id}] LLM 错误: {token}")
                break

            # 累加文本
            full_text += token

            # 句子分句
            sentences = self.splitter.feed(token)
            for sentence in sentences:
                if self.cancel_event.is_set():
                    break
                # 将句子加入文本队列
                await self.queues.text.put((seq_id, sentence))
                debug(f"[Session] 句子 #{seq_id} 入 text_queue: {sentence[:50]}...")
                seq_id += 1

        # 刷新剩余文本
        if not self.cancel_event.is_set():
            remaining = self.splitter.flush()
            if remaining:
                await self.queues.text.put((seq_id, remaining))
                seq_id += 1

    except StopPipeline:
        raise
    except Exception as e:
        error(f"[Session] LLM 任务异常: {e}")
    finally:
        # 发送结束标记
        self.queues.text.put_nowait((-1, None))
        info(f"[Session] LLM 结束，共 {seq_id} 句，full_text={len(full_text)}字符")

    # 更新对话记忆
    if memory_enabled and full_text:
        self.conversation_memory.add_user_message(iat_text)
        self.conversation_memory.add_assistant_message(full_text)

    return full_text
```

## ⚠️ 注意事项

### 1. 状态机状态一致性

确保每次状态切换都与发送的消息一致：

```python
# ✅ 正确
await fsm.set(SessionState.ASR)
await channel.send_json({"type": "session_status", "status": "iat_start"})

# ❌ 错误
await channel.send_json({"type": "session_status", "status": "iat_start"})
await fsm.set(SessionState.ASR)  # 顺序错误
```

### 2. 取消事件处理

在所有异步循环中检查取消事件：

```python
async for token in llm.stream_chat(messages):
    if self.cancel_event.is_set():
        break  # 必须检查取消信号
    yield token
```

### 3. 异常安全

使用 try-finally 确保资源释放：

```python
try:
    # 业务逻辑
    pass
finally:
    # 清理资源
    self.queues.text.put_nowait((-1, None))  # 发送结束标记
```
