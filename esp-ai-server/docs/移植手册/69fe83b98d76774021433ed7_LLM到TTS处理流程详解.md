## 📋 概述

本文档详细说明 LLM（大语言模型）处理完成后，如何将文本转换为语音（TTS）的完整流程，包括分句、合成、发送等环节。

### 步骤1: LLM 流式输出

**文件位置**: `app/session/session.py` 第374-421行

**函数**: `_llm_task()`

```python
async def _llm_task(self, llm, messages):
    seq_id = 0
    full_text = ""

    try:
        # 1.1 流式调用LLM
        async for token in llm.stream_chat(messages):
            # 检查取消信号
            if self.cancel_event.is_set():
                info(f"[Session] LLM 收到取消信号")
                break

            # 检查错误
            if token.startswith("LLM error"):
                error(f"[Session] LLM 错误: {token}")
                break

            # 累加完整文本
            full_text += token

            # 1.2 句子分句
            sentences = self.splitter.feed(token)

            # 1.3 将每个完整句子加入text队列
            for sentence in sentences:
                if self.cancel_event.is_set():
                    break
                await self.queues.text.put((seq_id, sentence))
                debug(f"[Session] 句子 #{seq_id} 入 text_queue: {sentence[:50]}...")
                seq_id += 1

        # 1.4 刷新剩余文本（最后一句话可能没有句号）
        if not self.cancel_event.is_set():
            remaining = self.splitter.flush()
            if remaining:
                await self.queues.text.put((seq_id, remaining))
                seq_id += 1

    except StopPipeline:
        raise
    except Exception as e:
        error(f"[Session] LLM 任务异常: {e}")
    finally:
        # 1.5 发送结束标记
        self.queues.text.put_nowait((-1, None))
        info(f"[Session] LLM 结束，共 {seq_id} 句，full_text={len(full_text)}字符")

    # 1.6 更新对话记忆
    if memory_enabled and full_text:
        self.conversation_memory.add_user_message(iat_text)
        self.conversation_memory.add_assistant_message(full_text)

    return full_text
```

**关键点**:
- 实时分句，边生成边分句
- 使用 `SentenceSplitter` 进行智能断句
- 将完整句子放入 `text_queue`

### 步骤2: 句子分句器

**文件位置**: `app/pipeline/sentence_splitter.py`

**类**: `SentenceSplitter`

```python
class SentenceSplitter:
    """句子分割器：将流式 token 组装成完整句子"""

    def __init__(self):
        self.buffer = ""

    def feed(self, token: str) -> list:
        """
        喂入一个 token，返回已完成的句子列表
        """
        if not token:
            return []

        # 累积到缓冲区
        self.buffer += token
        sentences = []

        # 检测断句符号
        if any(p in self.buffer for p in [".", "!", "?", "。", "！", "？"]):
            # 使用正则分割
            parts = re.split(r"([。！？.!?])", self.buffer)
            tmp = ""
            for p in parts:
                tmp += p
                if p in "。！？.!?":  # 遇到句号/问号/感叹号
                    stripped = tmp.strip()
                    if stripped and len(stripped) > 1:  # 过滤空句子
                        sentences.append(stripped)
                    tmp = ""  # 重置缓冲区
            self.buffer = tmp  # 保留未完成的部分

        return sentences

    def flush(self) -> str:
        """刷新剩余缓冲区内容（最后一句话）"""
        remaining = self.buffer.strip()
        self.buffer = ""
        return remaining if remaining and len(remaining) > 1 else ""

    def reset(self):
        """重置状态"""
        self.buffer = ""
```

**断句规则**:
- 中文句号: `。`
- 中文问号: `？`
- 中文感叹号: `！`
- 英文句号: `.`
- 英文问号: `?`
- 英文感叹号: `!`

**示例**:
```python
splitter = SentenceSplitter()

# 流式输入tokens
tokens = ["今天", "天气", "很好", "。", "我们", "去", "散步", "吧", "。"]

for token in tokens:
    sentences = splitter.feed(token)
    if sentences:
        print(f"完整句子: {sentences}")

# 处理最后可能没有句号的句子
remaining = splitter.flush()
if remaining:
    print(f"最后句子: {remaining}")
```

**输出**:
```
完整句子: ['今天天气很好。']
完整句子: ['我们去散步吧。']
```

### 步骤3: Splitter Worker

**文件位置**: `app/session/session.py` 第423-444行

**函数**: `_splitter_task()`

```python
async def _splitter_task(self):
    try:
        while True:
            # 检查取消信号
            if self.cancel_event.is_set():
                return

            # 从text_queue取出句子
            seq_id, sentence = await self.queues.text.get()
            try:
                # 3.1 收到结束标记
                if seq_id == -1 or sentence is None:
                    # 通知下游TTS结束
                    await self.queues.audio.put((-1, None, None))
                    return

                # 3.2 过滤空句子
                if not sentence.strip() or len(sentence.strip()) <= 1:
                    continue

                # 3.3 将句子放入audio_queue
                # 格式: (seq_id, sentence_text, sentence_text)
                await self.queues.audio.put((seq_id, sentence, sentence))

            finally:
                self.queues.text.task_done()

    except asyncio.CancelledError:
        pass
    except Exception as e:
        error(f"[Session] Splitter 任务异常: {e}")
```

**职责**:
- 从 `text_queue` 消费句子
- 过滤空句子和过短句子
- 传递给 `audio_queue`

### 步骤4: TTS Worker

**文件位置**: `app/session/session.py` 第446-511行

**函数**: `_tts_task()`

```python
async def _tts_task(self, volc_tts, session):
    # 4.1 判断是否需要创建TTS会话
    own_session = session is None
    if own_session:
        try:
            session = await volc_tts.create_session(cancel_event=self.cancel_event)
        except Exception as e:
            error(f"[Session] TTS 创建 session 失败: {e}")
            await self.queues.send.put((-1, None, None))
            return

    try:
        while True:
            # 检查取消信号
            if self.cancel_event.is_set():
                await self.queues.send.put((-1, None, None))
                return

            # 从audio_queue取出句子
            seq_id, sentence_text, _ = await self.queues.audio.get()

            # 再次检查取消
            if self.cancel_event.is_set():
                self.queues.audio.task_done()
                await self.queues.send.put((-1, None, None))
                return

            # 收到结束标记
            if seq_id == -1 or sentence_text is None:
                self.queues.audio.task_done()
                await self.queues.send.put((-1, None, None))
                return

            # 过滤空句子
            if not sentence_text.strip() or len(sentence_text.strip()) <= 1:
                self.queues.audio.task_done()
                continue

            # 4.2 情绪检测（可选）
            if SERVER_EMOTION_ENABLED:
                emotion = detect_emotion(sentence_text, self.device_id)
                if emotion and emotion != self._last_emotion:
                    self._last_emotion = emotion
                    asyncio.create_task(send_emotion_image(self.channel, sentence_text, self.device_id))

            info(f"[Session] TTS 合成 #{seq_id}: {sentence_text[:60]}...")

            # 4.3 调用TTS合成
            try:
                async with tts_semaphore:  # 并发控制
                    async for audio_chunk in session.synthesize(sentence_text, cancel_event=self.cancel_event):
                        if self.cancel_event.is_set():
                            break
                        if audio_chunk:
                            # 4.4 封装音频帧
                            frame = self.voice_generator.make_tts_frame(SID_TTS, audio_chunk, "00")
                            # 4.5 放入send_queue
                            await self.queues.send.put((seq_id, frame, sentence_text))
                            sentence_text = None  # 只发送一次文本
            except asyncio.CancelledError:
                self.queues.audio.task_done()
                raise
            except Exception as e:
                error(f"[Session] TTS 合成 #{seq_id} 异常: {e}")
                await self.queues.send.put((seq_id, b"", None))

            self.queues.audio.task_done()

    except asyncio.CancelledError:
        pass
    except Exception as e:
        error(f"[Session] TTS 任务异常: {e}")
    finally:
        # 关闭自己创建的session
        if own_session and session:
            try:
                await session.close()
            except Exception as e:
                debug(f"[Session] 关闭 TTS session 异常: {e}")
```

**关键点**:
- 使用 `tts_semaphore` 控制并发（默认10个）
- 调用火山引擎TTS进行流式合成
- 实时流式接收音频分片

### 步骤5: TTS 合成

**文件位置**: `app/tts/volcengine.py` 第284-408行

**类**: `TTSSession`

```python
class TTSSession:
    """复用的 TTS 会话：一个 WS 连接内合成多个句子"""

    def __init__(self, tts, websocket, session_id: str):
        self._tts = tts
        self._websocket = websocket
        self._session_id = session_id
        self._seq = 0
        self._closed = False

    async def synthesize(self, text: str, cancel_event=None):
        if self._closed:
            return

        # 限制文本长度
        if len(text) > 500:
            warning(f"[TTS] 文本长度过长: {len(text)}字符")

        self._seq += 1
        seq = self._seq

        # 5.1 构造请求
        request = {
            "user": {"uid": str(uuid.uuid4())},
            "req_params": {
                "speaker": self._tts.voice_type,  # 音色
                "audio_params": {
                    "format": "mp3",           # 音频格式
                    "sample_rate": 24000,      # 采样率
                    "speed_ratio": self._tts.speed_ratio,   # 语速
                    "volume_ratio": self._tts.volume_ratio, # 音量
                    "pitch_ratio": self._tts.pitch_ratio,   # 音调
                },
                "text": text,  # 要合成的文本
            },
        }

        try:
            # 5.2 发送合成请求
            await full_client_request(self._websocket, json.dumps(request).encode())
        except Exception as e:
            # 重连逻辑...
            ...

        try:
            # 5.3 流式接收音频
            while True:
                if cancel_event and cancel_event.is_set():
                    info(f"[TTS] 合成 #{seq} 被取消信号中断")
                    return

                # 接收消息
                msg = await asyncio.wait_for(receive_message(self._websocket), timeout=20)

                # 5.4 处理音频数据
                if msg.type == MsgType.AudioOnlyServer:
                    if msg.payload:
                        yield msg.payload  # 流式返回音频分片

                # 5.5 处理会话结束
                elif msg.type == MsgType.FullServerResponse:
                    if msg.event == EventType.SessionFinished:
                        debug(f"[TTS] 句子 #{seq} 合成完成")
                        return

        except (websockets.exceptions.ConnectionClosed, OSError, ConnectionError) as e:
            error(f"[TTS] WS 连接异常 #{seq}: {e}")
            self._closed = True
```

**TTS请求参数**:

```python
{
    "user": {
        "uid": "唯一标识符"
    },
    "req_params": {
        "speaker": "zh_female_wanwanxiaohe_moon_bigtts",  # 音色
        "audio_params": {
            "format": "mp3",       # 音频格式: mp3
            "sample_rate": 24000, # 采样率: 24kHz
            "speed_ratio": 1.0,   # 语速: 0.5~2.0
            "volume_ratio": 1.0,   # 音量: 0.5~2.0
            "pitch_ratio": 1.0,    # 音调: 0.5~2.0
        },
        "text": "要合成的文本"
    }
}
```

**TTS响应类型**:

| 消息类型 | 说明 |
|----------|------|
| `AudioOnlyServer` | 音频数据分片 |
| `FullServerResponse` + `SessionFinished` | 句子合成完成 |

### 步骤6: 音频帧封装

**文件位置**: `app/tts/voice_generator.py`

**类**: `VoiceGenerator`

```python
class VoiceGenerator:
    def make_tts_frame(self, session_id, audio_data, status="00"):
        """
        创建 TTS 数据帧
        帧结构：
        - 会话ID (4字节 utf-8)
        - 会话状态 (2字节 utf-8)
        - 原始音频数据
        """
        sid = session_id.encode("utf-8")   # b"0010"
        ss = status.encode("utf-8")        # b"00"
        return sid + ss + audio_data       # 拼接

    def make_end_frame(self, session_id):
        """
        创建结束帧
        帧结构：会话ID(4字节) + 状态03(2字节)
        """
        sid = session_id.encode("utf-8")
        return sid + b"03"
```

**帧格式**:

```
┌────────────────┬───────────┬─────────────────┐
│   session_id   │   status  │   audio_data   │
│   (4 bytes)    │ (2 bytes) │   (variable)   │
├────────────────┼───────────┼─────────────────┤
│     "0010"     │   "00"    │   MP3音频数据   │
└────────────────┴───────────┴─────────────────┘

状态码:
- "00": 正常音频帧
- "03": 结束帧
```

### 步骤7: Sender Worker

**文件位置**: `app/session/session.py` 第513-592行

**函数**: `_sender_task()`

```python
async def _sender_task(self):
    total_sent = 0
    frame_count = 0
    current_seq_id = -1
    pending_text = None
    text_delay_frames = 0
    text_send_delay = 8  # 延迟8帧后发送文本

    try:
        while True:
            # 检查取消信号
            if self.cancel_event.is_set():
                return

            # 从send_queue取出帧
            seq_id, audio, sentence_text = await self.queues.send.get()

            # 取消检查
            if self.cancel_event.is_set():
                self.queues.send.task_done()
                return

            # 收到结束标记
            if seq_id == -1 or audio is None:
                info(f"[Session] Sender 结束，{frame_count}帧/{total_sent}B")
                self.queues.send.task_done()
                return

            if audio:
                # 取消检查
                if self.cancel_event.is_set():
                    self.queues.send.task_done()
                    return

                try:
                    # 7.1 检测句子切换
                    if seq_id != current_seq_id:
                        current_seq_id = seq_id
                        if sentence_text:
                            pending_text = sentence_text
                            text_delay_frames = 0

                    # 7.2 延迟发送文本
                    if pending_text is not None:
                        text_delay_frames += 1
                        if text_delay_frames >= text_send_delay:
                            if self.cancel_event.is_set():
                                pending_text = None
                                self.queues.send.task_done()
                                return
                            # 发送LLM回复文本回调
                            try:
                                await self.channel.send_json(
                                    {"type": "instruct", "command_id": "on_llm_cb", "data": pending_text}
                                )
                            except Exception as e:
                                debug(f"[Session] 发送 instruct 失败: {e}")
                            pending_text = None
                            text_delay_frames = 0

                    # 取消检查
                    if self.cancel_event.is_set():
                        self.queues.send.task_done()
                        return

                    # 7.3 发送音频帧
                    await self.channel.send_bytes(audio)
                    total_sent += len(audio)
                    frame_count += 1

                    # 每50帧打印日志
                    if frame_count % 50 == 0:
                        info(f"[Session] Sender: {frame_count}帧/{total_sent}B")

                except Exception as e:
                    debug(f"[Session] Sender 发送失败: {e}")
                    self.queues.send.task_done()
                    return

                # 7.4 流量控制
                if self._device_buffer > client_max_buffer * 0.8:
                    await asyncio.sleep(0.2)  # 设备缓冲区满，减速
                else:
                    await asyncio.sleep(0.1)  # 正常速度

            self.queues.send.task_done()

    except asyncio.CancelledError:
        pass
    except Exception as e:
        error(f"[Session] Sender 异常: {e}")
```

**关键逻辑**:
- 延迟8帧后发送 `instruct: on_llm_cb`（确保音频先开始播放）
- 流量控制：根据设备缓冲区状态调整发送速度
- 异常处理和取消检查

## 🔄 Pipeline 并行执行

**文件位置**: `app/session/session.py` 第310-353行

```python
async def run_pipeline(self, iat_text: str):
    # 初始化
    self.cancel_event.clear()
    self.tts_playback_done.clear()
    self.tts_drain_ack.clear()
    self.tts_audio_ended.clear()
    self.queues.clear_all()
    self.splitter.reset()

    await self.set_tts_playing(True)
    await self.fsm.set(SessionState.LLM)

    # 构建消息
    messages = self.conversation_memory.build_messages(system_prompt, iat_text)

    # 切换状态
    await self.fsm.set(SessionState.TTS)

    # 通知客户端开始播放
    await self.channel.send_json({"type": "play_audio", "tts_task_id": SID_TTS})
    await asyncio.sleep(0.03)
    await self.channel.send_json({"type": "session_status", "status": "tts_chunk_start"})

    # 创建TTS会话
    volc_tts = self.tts_processor
    session = await volc_tts.create_session(cancel_event=self.cancel_event)

    # 启动4个Worker
    tasks = []
    t_llm = asyncio.create_task(self._llm_task(llm, messages))
    t_splitter = asyncio.create_task(self._splitter_task())
    t_tts = asyncio.create_task(self._tts_task(volc_tts, session))
    t_sender = asyncio.create_task(self._sender_task())
    tasks = [t_llm, t_splitter, t_tts, t_sender]

    info(f"[Session] Pipeline 已启动 (4 workers)")

    # 等待任意Worker异常或全部完成
    done, pending = await asyncio.wait(
        tasks, return_when=asyncio.FIRST_EXCEPTION
    )

    # 取消所有Pending任务
    for t in pending:
        t.cancel()

    # 处理异常
    for t in done:
        exc = t.exception()
        if exc and not isinstance(exc, (asyncio.CancelledError, StopPipeline)):
            error(f"[Session] Worker 异常: {exc}")

    # 清理Pending任务
    for t in pending:
        try:
            await t
        except (asyncio.CancelledError, StopPipeline):
            pass
        except Exception as e:
            error(f"[Session] Worker 清理异常: {e}")
```

## 📊 背压队列机制

**文件位置**: `app/session/queues.py`

```python
class BackpressureQueues:
    def __init__(self):
        # text_queue: 句子队列 (maxsize=10, drop_oldest)
        self.text = TextQueue(maxsize=10)

        # audio_queue: 音频队列 (maxsize=20, block)
        self.audio = AudioQueue(maxsize=20)

        # send_queue: 发送队列 (maxsize=50, block)
        self.send = SendQueue(maxsize=50)

    def clear_all(self):
        self.text.clear()
        self.audio.clear()
        self.send.clear()
```

**队列策略**:

| 队列 | 最大容量 | 满时策略 | 说明 |
|------|----------|----------|------|
| `text_queue` | 10 | `drop_oldest` | 丢弃最旧的句子 |
| `audio_queue` | 20 | `block` | 阻塞等待 |
| `send_queue` | 50 | `block` | 阻塞等待 |

**背压机制**:

```
LLM生成太快 ──► text_queue满 ──► 丢弃旧句子 ──► LLM继续生成

TTS处理太慢 ──► audio_queue满 ──► LLM/Splitter阻塞 ──► LLM暂停生成

网络发送慢 ──► send_queue满 ──► TTS阻塞 ──► audio_queue积压
```

## ⚠️ 注意事项

### 1. 取消信号传播

在所有异步循环中检查取消信号：

```python
if self.cancel_event.is_set():
    return  # 必须检查
```

### 2. 队列阻塞

避免在关键路径上长时间阻塞：

```python
# 不好: 阻塞等待
await self.queues.audio.get()

# 好: 设置超时
try:
    item = await asyncio.wait_for(self.queues.audio.get(), timeout=1.0)
except asyncio.TimeoutError:
    continue
```

### 3. 异常安全

确保资源正确释放：

```python
try:
    session = await volc_tts.create_session()
    async for chunk in session.synthesize(text):
        yield chunk
finally:
    await session.close()  # 必须关闭
```

## 📚 参考代码

### 完整Pipeline启动示例

```python
async def run_pipeline(self, iat_text: str):
    # 1. 初始化
    self.cancel_event.clear()
    self.queues.clear_all()
    self.splitter.reset()

    # 2. 通知客户端
    await self.channel.send_json({"type": "play_audio", "tts_task_id": SID_TTS})
    await self.channel.send_json({"type": "session_status", "status": "tts_chunk_start"})

    # 3. 创建TTS会话
    session = await self.tts_processor.create_session()

    # 4. 启动4个Worker
    t_llm = asyncio.create_task(self._llm_task(self.llm_processor, messages))
    t_splitter = asyncio.create_task(self._splitter_task())
    t_tts = asyncio.create_task(self._tts_task(self.tts_processor, session))
    t_sender = asyncio.create_task(self._sender_task())

    # 5. 等待完成
    await asyncio.gather(t_llm, t_splitter, t_tts, t_sender)

    # 6. 发送结束
    await self.channel.send_bytes(self.voice_generator.make_end_frame(SID_TTS))
    await self.channel.send_json({"type": "session_status", "status": "tts_real_end"})
```
